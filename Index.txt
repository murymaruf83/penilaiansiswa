        function prosesUploadSiswa(e) {
            e.preventDefault();
            const fileInput = document.getElementById('fileUploadSiswa');
            const file = fileInput.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                const text = event.target.result;
                const lines = text.split('\n');
                const dataSiswaBaru = [];
                
                // Mulai dari 1 untuk melewati baris Header
                for(let i = 1; i < lines.length; i++) {
                    const line = lines[i].trim();
                    if(line) {
                        const parts = line.split(',');
                        if(parts.length >= 3) {
                            dataSiswaBaru.push({
                                nisn: parts[0].trim(),
                                nama: parts[1].trim(),
                                kelas: parts[2].trim()
                            });
                        }
                    }
                }

                if(dataSiswaBaru.length === 0) {
                    showToast('Data CSV kosong atau format salah', 'error');
                    return;
                }

                const btn = document.getElementById('btnUploadSiswa');
                btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Menyimpan...';
                btn.disabled = true;

                if (IS_PREVIEW) {
                    setTimeout(() => {
                        dataSiswaBaru.forEach(s => dummySiswa.push(s));
                        showToast(`${dataSiswaBaru.length} Siswa berhasil diupload!`, 'success');
                        tutupModalUploadSiswa();
                        loadSiswaList();
                        btn.innerHTML = 'Upload & Simpan';
                        btn.disabled = false;
                    }, 1000);
                } else {
                    google.script.run.withSuccessHandler((res) => {
                        showToast(`${dataSiswaBaru.length} Siswa berhasil diupload!`, 'success');
                        tutupModalUploadSiswa();
                        loadSiswaList();
                        btn.innerHTML = 'Upload & Simpan';
                        btn.disabled = false;
                    }).tambahSiswaBulk(dataSiswaBaru);
                }
            };
            reader.readAsText(file);
        }

        function hitungMaxTP(nilaiMap) {
            currentMaxTP = 1; 
            let hasData = false;
            for (const nisn in nilaiMap) {
                hasData = true;
                const tpData = nilaiMap[nisn];
                for (const key in tpData) {
                    if (key.startsWith('tp')) {
                        const num = parseInt(key.replace('tp', ''));
                        if (num > currentMaxTP) currentMaxTP = num;
                    }
                }
            }
            if(!hasData) currentMaxTP = 3;
        }

        function calcTP(rowIdx, tpNum) {
            const sStr = document.getElementById(`s_${rowIdx}_${tpNum}`).value;
            const rStr = document.getElementById(`r_${rowIdx}_${tpNum}`).value;
            const pStr = document.getElementById(`p_${rowIdx}_${tpNum}`).value; 
            
            const kktpEl = document.getElementById(`kktp_${tpNum}`);
            const kktp = kktpEl ? parseFloat(kktpEl.value) || 0 : 0;
            
            const s = sStr !== '' ? parseFloat(sStr) : 0;
            const r = rStr !== '' ? parseFloat(rStr) : 0;
            const p = pStr !== '' ? parseFloat(pStr) : 0;
            
            let nilai = Math.max(s, r);
            
            // 1. ATURAN REMEDIAL (Mentok di KKTP)
            if (s > 0 && s < kktp) {
                if (nilai > kktp) nilai = kktp; 
            }
            
            // 2. ATURAN PENGAYAAN
            if (s >= kktp && p > 0) {
                if (p > s) {
                    nilai = p;
                } else {
                    nilai = s + (s * 0.1); 
                }
            }
            
            if (nilai > 100) nilai = 100;
            nilai = Math.round(nilai);
            
            const nSpan = document.getElementById(`n_${rowIdx}_${tpNum}`);
            const tdNilai = document.getElementById(`td_n_${rowIdx}_${tpNum}`);
            nSpan.innerText = nilai;
            
            if(nilai > 0) {
                tdNilai.classList.remove('text-red-500', 'bg-white', 'text-slate-800', 'text-red-600');
                tdNilai.classList.add('bg-yellow-200');
                if (nilai < kktp) {
                    tdNilai.classList.add('text-red-600'); 
                } else {
                    tdNilai.classList.add('text-slate-800'); 
                }
            } else {
                tdNilai.classList.remove('bg-yellow-200', 'text-slate-800', 'text-red-600');
                tdNilai.classList.add('text-red-500', 'bg-white');
            }
            
            // Panggil update Rata-rata secara real-time
            updateRataRataBaris(rowIdx);
        }

        // Fungsi real-time hitung Rata-rata dan Bobot Nilai Akhir
        function updateRataRataBaris(rowIdx) {
            let total = 0;
            let count = 0;
            for(let i = 1; i <= currentMaxTP; i++) {
                const nSpan = document.getElementById(`n_${rowIdx}_${i}`);
                if(nSpan) {
                    const val = parseInt(nSpan.innerText) || 0;
                    if(val > 0) { total += val; count++; }
                }
            }
            
            const rata = count > 0 ? Math.round(total / count) : 0;
            const rataSpan = document.getElementById(`rata_${rowIdx}`);
            if(rataSpan) rataSpan.innerText = rata;

            // Kalkulasi Real-time Nilai Rapor Asli (70% Rata-rata TP + 30% Sumatif Akhir)
            let nr = rata;
            const saInput = document.getElementById(`sa_${rowIdx}`);
            if (saInput && saInput.value !== '' && !isNaN(saInput.value)) {
                nr = Math.round((rata * 0.7) + (parseFloat(saInput.value) * 0.3));
            }
            const nrSpan = document.getElementById(`nr_${rowIdx}`);
            if (nrSpan) nrSpan.innerText = nr;

            // Update Predikat secara Real-time dengan warnanya
            const predSpan = document.getElementById(`pred_${rowIdx}`);
            const tdPred = document.getElementById(`td_pred_${rowIdx}`);
            if (predSpan && tdPred) {
                const pred = hitungPredikat(nr);
                predSpan.innerText = pred;
                tdPred.className = `border border-slate-300 bg-orange-50 p-1 font-bold text-base ${pred === 'A' ? 'text-green-600' : (pred === 'B' ? 'text-blue-600' : (pred === 'C' ? 'text-yellow-600' : 'text-red-600'))}`;
            }
        }

  for (let i = 1; i < data.length; i++) {

        status: 'success',

  return { status: 'failed' };

/**
 * 5. Fungsi Admin: Simpan Nilai Bulk (Grid Mode)
 */
function simpanNilaiGridBulk(kelas, mapel, dataGrid) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Nilai');
  let data = sheet.getDataRange().getValues();
  const timestamp = new Date();
  
  const mapBaris = {};
  for (let i = 1; i < data.length; i++) {
    const nisnSheet = data[i][1].toString();
    const mapelSheet = data[i][2].toString();
    mapBaris[nisnSheet + "_" + mapelSheet] = i + 1; // row index
  }

  dataGrid.forEach(item => {
    const key = item.nisn + "_" + mapel;
    const detailJSON = JSON.stringify(item.tpData);
    
    if (mapBaris[key]) {
      // Update
      sheet.getRange(mapBaris[key], 1, 1, 7).setValues([[
        timestamp, item.nisn, mapel, kelas, item.akhir, item.predikat, detailJSON
      ]]);
    } else {
      // Insert
      sheet.appendRow([
        timestamp, item.nisn, mapel, kelas, item.akhir, item.predikat, detailJSON
      ]);
    }
  });
  
  return true;
}

/**
 * 5b. Ambil Data Grid (Siswa + Nilai)
 */
function getSiswaAndNilaiGrid(kelas, mapel) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Ambil Siswa
  const sheetSiswa = ss.getSheetByName('Siswa');
  const dataSiswa = sheetSiswa.getDataRange().getValues();
  const listSiswa = [];
  const nisnSet = new Set();
  
  for (let i = 1; i < dataSiswa.length; i++) {
    if (dataSiswa[i][2] === kelas) {
      listSiswa.push({ nisn: dataSiswa[i][0].toString(), nama: dataSiswa[i][1] });
      nisnSet.add(dataSiswa[i][0].toString());
    }
  }

  // Ambil Nilai
  const sheetNilai = ss.getSheetByName('Nilai');
  const dataNilai = sheetNilai.getDataRange().getValues();
  const mapNilai = {};
  
  for (let i = 1; i < dataNilai.length; i++) {
    const nisn = dataNilai[i][1].toString();
    const mpl = dataNilai[i][2].toString();
    if (nisnSet.has(nisn) && mpl === mapel) {
      try {
        mapNilai[nisn] = JSON.parse(dataNilai[i][6]); // DetailJSON dari kolom 7
      } catch(e) {
        mapNilai[nisn] = {};
      }
    }
  }

  return { siswa: listSiswa, nilaiMap: mapNilai };
}

/**
 * 6. Fungsi Admin: Ambil Rekap Kelas
 */
function getRekapKelas(kelas) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Ambil data siswa di kelas tsb
  const sheetSiswa = ss.getSheetByName('Siswa');
  const dataSiswa = sheetSiswa.getDataRange().getValues();
  const listSiswa = [];
  const nisnSet = new Set();
  
  for (let i = 1; i < dataSiswa.length; i++) {
    if (dataSiswa[i][2] === kelas) {
      listSiswa.push({ nisn: dataSiswa[i][0].toString(), nama: dataSiswa[i][1], kelas: dataSiswa[i][2] });
      nisnSet.add(dataSiswa[i][0].toString());
    }
  }
  
  // 2. Ambil nilai hanya untuk siswa di kelas tersebut
  const sheetNilai = ss.getSheetByName('Nilai');
  const dataNilai = sheetNilai.getDataRange().getValues();
  const listNilai = [];
  
  for (let i = 1; i < dataNilai.length; i++) {
    const nisnSheet = dataNilai[i][1].toString();
    if (nisnSet.has(nisnSheet)) {
      listNilai.push({
        nisn: nisnSheet,
        mapel: dataNilai[i][2],
        akhir: dataNilai[i][4],     // NilaiAkhir di col index 4
        predikat: dataNilai[i][5]   // Predikat di col index 5
      });
    }
  }
  
  return { siswa: listSiswa, nilai: listNilai };
}

 * 7. Fungsi Siswa: Ambil Data Rapor Pribadi
 */
function getDataSiswa(nisnStr) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Ambil profil
  const sheetSiswa = ss.getSheetByName('Siswa');
  const dataSiswa = sheetSiswa.getDataRange().getValues();
  let profil = { nisn: nisnStr, nama: 'Tidak Ditemukan', kelas: '-' };
  
  for (let i = 1; i < dataSiswa.length; i++) {
    if (dataSiswa[i][0].toString() === nisnStr) {
      profil = { nisn: nisnStr, nama: dataSiswa[i][1], kelas: dataSiswa[i][2] };
      break;
    }
  }
  
  // Ambil Nilai
  const sheetNilai = ss.getSheetByName('Nilai');
  const dataNilai = sheetNilai.getDataRange().getValues();
  const listNilai = [];
  
  for (let i = 1; i < dataNilai.length; i++) {
    if (dataNilai[i][1].toString() === nisnStr) {
      let tpData = {};
      try { tpData = JSON.parse(dataNilai[i][6]); } catch(e){}
      listNilai.push({
        mapel: dataNilai[i][2],
        formatif: '-', // Tidak dipakai lagi di format baru
        sumatif: '-',
        akhir: dataNilai[i][4],
        predikat: dataNilai[i][5],

  }
  
  return { profil: profil, nilai: listNilai };
}
